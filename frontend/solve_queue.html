Junior's Query

"I'm trying to implement Dijkstra's for my algorithms assignment, but it's failing the time complexity tests for dense graphs. I used a standard list for the priority queue. How can I optimize this in Python without using external libraries?"

description Detailed Description

def dijkstra(graph, start):
    distances = {node: float('infinity') for node in graph}
    distances[start] = 0
    unvisited = list(graph.keys())
    
    while unvisited:
        # This is where I think the bottleneck is
        current_node = min(unvisited, key=lambda node: distances[node])
        unvisited.remove(current_node)
        
        for neighbor, weight in graph[current_node].items():
            new_distance = distances[current_node] + weight
            if new_distance < distances[neighbor]:
                distances[neighbor] = new_distance
                
    return distances
